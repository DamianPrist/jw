```python
"""局域网即时通信系统 — 共享协议层"""

import json
from typing import Any

# 消息类型常量
TYPE_LOGIN = "login"
TYPE_LOGIN_OK = "login_ok"
TYPE_LOGIN_FAIL = "login_fail"
TYPE_LOGOUT = "logout"
TYPE_BROADCAST = "broadcast"
TYPE_PRIVATE = "private"
TYPE_USER_ONLINE = "user_online"
TYPE_USER_OFFLINE = "user_offline"
TYPE_ERROR = "error"
TYPE_DISCONNECTED = "disconnected"

# 限制
MAX_MESSAGE_LENGTH = 64 * 1024  # 64KB
NICKNAME_MAX_LENGTH = 20
FORBIDDEN_NICKNAME_CHARS = set(" @\n\r\t")

ENCODING = "utf-8"
DELIMITER = b"\n"


def is_valid_nickname(nickname: str) -> tuple[bool, str]:
    """校验昵称合法性。返回 (是否合法, 错误原因)。"""
    if not nickname:
        return False, "昵称不能为空"
    if len(nickname) > NICKNAME_MAX_LENGTH:
        return False, f"昵称不能超过{NICKNAME_MAX_LENGTH}个字符"
    for ch in nickname:
        if ch in FORBIDDEN_NICKNAME_CHARS:
            return False, "昵称不能包含空格、@、换行等特殊字符"
    return True, ""


def pack_message(msg: dict[str, Any]) -> bytes:
    """将消息字典序列化为 JSON 并追加换行分隔符。"""
    body = json.dumps(msg, ensure_ascii=False)
    if len(body) > MAX_MESSAGE_LENGTH:
        raise ValueError(f"消息过长: {len(body)} bytes")
    return body.encode(ENCODING) + DELIMITER


def unpack_from_buffer(buffer: bytearray) -> list[dict[str, Any]]:
    """从接收缓冲区中提取所有完整的 JSON 消息。返回解析后的消息列表。"""
    messages = []
    while True:
        idx = buffer.find(DELIMITER)
        if idx == -1:
            break
        raw = bytes(buffer[:idx])
        del buffer[:idx + 1]
        if raw:
            try:
                msg = json.loads(raw.decode(ENCODING))
                messages.append(msg)
            except (json.JSONDecodeError, UnicodeDecodeError):
                pass  # 丢弃无法解析的数据
    return messages
```

```python
"""局域网即时通信系统 — 服务端"""

import socket
import select
import sys
from protocol import (
    pack_message, unpack_from_buffer, is_valid_nickname,
    TYPE_LOGIN, TYPE_LOGIN_OK, TYPE_LOGIN_FAIL, TYPE_LOGOUT,
    TYPE_BROADCAST, TYPE_PRIVATE, TYPE_USER_ONLINE, TYPE_USER_OFFLINE,
    TYPE_ERROR,
)


def broadcast(serversock, clients, msg, exclude_sock=None):
    """向所有已登录客户端广播消息，可排除指定 socket。"""
    data = pack_message(msg)
    for sock in clients:
        if sock is not exclude_sock and sock is not serversock:
            try:
                sock.sendall(data)
            except OSError:
                pass


def handle_login(sock, msg, clients, nick_to_sock, buffers):
    """处理登录请求。"""
    nickname = msg.get("nickname", "").strip()
    valid, reason = is_valid_nickname(nickname)
    if not valid:
        sock.sendall(pack_message({"type": TYPE_LOGIN_FAIL, "reason": reason}))
        return

    if nickname in nick_to_sock:
        sock.sendall(pack_message({"type": TYPE_LOGIN_FAIL, "reason": "昵称已存在"}))
        return

    clients[sock] = nickname
    nick_to_sock[nickname] = sock

    online_users = list(nick_to_sock.keys())
    sock.sendall(pack_message({"type": TYPE_LOGIN_OK, "users": online_users}))
    broadcast(None, clients, {"type": TYPE_USER_ONLINE, "nickname": nickname}, exclude_sock=sock)
    print(f"[+] {nickname} 上线，当前在线: {online_users}")


def handle_broadcast(sock, msg, clients, serversock):
    """处理公聊消息转发。"""
    sender = clients.get(sock)
    if not sender:
        return
    forward = {"type": TYPE_BROADCAST, "from": sender, "content": msg.get("content", "")}
    broadcast(serversock, clients, forward, exclude_sock=sock)
    print(f"[公聊] {sender}: {msg.get('content', '')}")


def handle_private(sock, msg, clients, nick_to_sock):
    """处理私聊消息转发。"""
    sender = clients.get(sock)
    if not sender:
        return
    target = msg.get("to", "")
    content = msg.get("content", "")

    target_sock = nick_to_sock.get(target)
    if target_sock is None:
        sock.sendall(pack_message({"type": TYPE_ERROR, "reason": f"用户 {target} 不在线"}))
        return

    forward = {"type": TYPE_PRIVATE, "from": sender, "content": content}
    try:
        target_sock.sendall(pack_message(forward))
    except OSError:
        sock.sendall(pack_message({"type": TYPE_ERROR, "reason": f"发送失败"}))
    print(f"[私聊] {sender} -> {target}: {content}")


def disconnect_sock(sock, clients, nick_to_sock, buffers):
    """断开连接并清理资源。"""
    nickname = clients.pop(sock, None)
    if nickname:
        del nick_to_sock[nickname]
        broadcast(None, clients, {"type": TYPE_USER_OFFLINE, "nickname": nickname})
        print(f"[-] {nickname} 下线")
    buffers.pop(sock, None)
    try:
        sock.close()
    except OSError:
        pass


def main(host="0.0.0.0", port=9999):
    serversock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    serversock.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    serversock.bind((host, port))
    serversock.listen(10)
    serversock.setblocking(False)
    print(f"服务端启动: {host}:{port}")

    clients = {}       # {socket: nickname}
    nick_to_sock = {}  # {nickname: socket}
    buffers = {}       # {socket: bytearray}  每个连接的接收缓冲区

    while True:
        try:
            readable, _, _ = select.select(
                [serversock] + list(clients.keys()), [], []
            )
        except (select.error, OSError):
            continue
        except KeyboardInterrupt:
            print("\n服务端关闭")
            break

        for sock in readable:
            if sock is serversock:
                conn, addr = sock.accept()
                conn.setblocking(False)
                buffers[conn] = bytearray()
                print(f"[连接] {addr}")
                continue

            buffers.setdefault(sock, bytearray())

            try:
                data = sock.recv(4096)
            except (ConnectionResetError, ConnectionAbortedError, OSError):
                data = b""

            if not data:
                disconnect_sock(sock, clients, nick_to_sock, buffers)
                continue

            buffers[sock].extend(data)
            messages = unpack_from_buffer(buffers[sock])

            for msg in messages:
                msg_type = msg.get("type", "")

                if sock not in clients:
                    # 未登录：只接受 login
                    if msg_type == TYPE_LOGIN:
                        handle_login(sock, msg, clients, nick_to_sock, buffers)
                    continue

                if msg_type == TYPE_LOGIN:
                    continue  # 已登录忽略重复 login
                elif msg_type == TYPE_LOGOUT:
                    disconnect_sock(sock, clients, nick_to_sock, buffers)
                elif msg_type == TYPE_BROADCAST:
                    handle_broadcast(sock, msg, clients, serversock)
                elif msg_type == TYPE_PRIVATE:
                    handle_private(sock, msg, clients, nick_to_sock)

    for sock in list(clients.keys()):
        disconnect_sock(sock, clients, nick_to_sock, buffers)
    serversock.close()


if __name__ == "__main__":
    port = int(sys.argv[1]) if len(sys.argv) > 1 else 9999
    main(port=port)
```

```python
"""局域网即时通信系统 — 客户端"""

import socket
import threading
import queue
import tkinter as tk
from tkinter import messagebox, scrolledtext

from protocol import (
    pack_message, unpack_from_buffer, is_valid_nickname,
    TYPE_LOGIN, TYPE_LOGIN_OK, TYPE_LOGIN_FAIL, TYPE_LOGOUT,
    TYPE_BROADCAST, TYPE_PRIVATE, TYPE_USER_ONLINE, TYPE_USER_OFFLINE,
    TYPE_ERROR, TYPE_DISCONNECTED,
)


class LoginWindow:
    """登录窗口：IP、端口、昵称 + 连接按钮。"""

    def __init__(self):
        self.root = tk.Tk()
        self.root.title("登录 — LAN Chat")
        self.root.resizable(False, False)
        self.chat_window = None
        self.sock = None

        frame = tk.Frame(self.root, padx=20, pady=15)
        frame.pack()

        tk.Label(frame, text="服务器 IP:").grid(row=0, column=0, sticky="e", pady=4)
        self.ip_entry = tk.Entry(frame, width=22)
        self.ip_entry.insert(0, "127.0.0.1")
        self.ip_entry.grid(row=0, column=1, pady=4)

        tk.Label(frame, text="端口:").grid(row=1, column=0, sticky="e", pady=4)
        self.port_entry = tk.Entry(frame, width=22)
        self.port_entry.insert(0, "9999")
        self.port_entry.grid(row=1, column=1, pady=4)

        tk.Label(frame, text="昵称:").grid(row=2, column=0, sticky="e", pady=4)
        self.nick_entry = tk.Entry(frame, width=22)
        self.nick_entry.grid(row=2, column=1, pady=4)

        self.connect_btn = tk.Button(frame, text="连接", width=18, command=self.do_connect)
        self.connect_btn.grid(row=3, column=1, pady=10)

        self.nick_entry.bind("<Return>", lambda e: self.do_connect())

    def do_connect(self):
        nickname = self.nick_entry.get().strip()
        valid, reason = is_valid_nickname(nickname)
        if not valid:
            messagebox.showwarning("昵称无效", reason)
            return

        host = self.ip_entry.get().strip()
        try:
            port = int(self.port_entry.get().strip())
        except ValueError:
            messagebox.showwarning("端口错误", "端口必须是整数")
            return

        try:
            self.sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            self.sock.settimeout(5)
            self.sock.connect((host, port))
            self.sock.settimeout(None)
        except (ConnectionRefusedError, socket.timeout, OSError) as e:
            messagebox.showerror("连接失败", f"无法连接到服务器\n{e}")
            if self.sock:
                self.sock.close()
                self.sock = None
            return

        # 发送登录请求
        self.sock.sendall(pack_message({"type": TYPE_LOGIN, "nickname": nickname}))

        # 等待登录响应（阻塞一小段时间）
        self.sock.settimeout(3)
        try:
            data = self.sock.recv(4096)
        except (socket.timeout, OSError):
            messagebox.showerror("登录超时", "服务器无响应")
            self.sock.close()
            self.sock = None
            return
        self.sock.settimeout(None)

        buf = bytearray(data)
        msgs = unpack_from_buffer(buf)
        if not msgs:
            messagebox.showerror("登录失败", "服务器返回无效数据")
            self.sock.close()
            self.sock = None
            return

        resp = msgs[0]
        if resp.get("type") == TYPE_LOGIN_OK:
            online_users = resp.get("users", [])
            self.launch_chat(nickname, online_users)
        elif resp.get("type") == TYPE_LOGIN_FAIL:
            messagebox.showerror("登录失败", resp.get("reason", "未知原因"))
            self.sock.close()
            self.sock = None
        else:
            messagebox.showerror("登录失败", "未知响应")
            self.sock.close()
            self.sock = None

    def launch_chat(self, nickname, online_users):
        self.root.withdraw()
        self.chat_window = ChatWindow(
            self.sock, nickname, online_users,
            on_close=self.on_chat_closed
        )

    def on_chat_closed(self):
        self.root.destroy()

    def run(self):
        self.root.mainloop()


class ChatWindow:
    """聊天窗口：消息显示 + 在线用户列表 + 输入框。"""

    def __init__(self, sock, nickname, online_users, on_close):
        self.sock = sock
        self.nickname = nickname
        self.on_close = on_close
        self.msg_queue = queue.Queue()
        self.running = True

        self.root = tk.Tk()
        self.root.title(f"LAN Chat — {nickname}")
        self.root.protocol("WM_DELETE_WINDOW", self.do_close)

        # --- 左侧在线用户列表 ---
        left_frame = tk.LabelFrame(self.root, text="在线用户", padx=4, pady=4)
        left_frame.pack(side=tk.LEFT, fill=tk.Y, padx=4, pady=4)

        self.user_listbox = tk.Listbox(left_frame, width=16, height=20)
        self.user_listbox.pack(fill=tk.BOTH, expand=True)
        for user in online_users:
            self.user_listbox.insert(tk.END, user)

        # --- 右侧消息区 ---
        right_frame = tk.Frame(self.root)
        right_frame.pack(side=tk.RIGHT, fill=tk.BOTH, expand=True, padx=4, pady=4)

        self.msg_display = scrolledtext.ScrolledText(
            right_frame, width=50, height=20, state=tk.DISABLED, wrap=tk.WORD
        )
        self.msg_display.pack(fill=tk.BOTH, expand=True)

        # --- 底部输入区 ---
        bottom_frame = tk.Frame(self.root)
        bottom_frame.pack(side=tk.BOTTOM, fill=tk.X, padx=4, pady=4)

        self.input_entry = tk.Entry(bottom_frame)
        self.input_entry.pack(side=tk.LEFT, fill=tk.X, expand=True)
        self.input_entry.bind("<Return>", lambda e: self.send_message())
        self.input_entry.focus_set()

        send_btn = tk.Button(bottom_frame, text="发送", width=8, command=self.send_message)
        send_btn.pack(side=tk.RIGHT, padx=(4, 0))

        # 启动接收线程
        self.recv_thread = threading.Thread(target=self.receive_loop, daemon=True)
        self.recv_thread.start()

        # 定时轮询队列
        self.poll_queue()

    def append_message(self, text):
        """在消息显示区追加一行文本。"""
        self.msg_display.config(state=tk.NORMAL)
        self.msg_display.insert(tk.END, text + "\n")
        self.msg_display.see(tk.END)
        self.msg_display.config(state=tk.DISABLED)

    def send_message(self):
        content = self.input_entry.get()
        if not content:
            return
        self.input_entry.delete(0, tk.END)

        if content.startswith("@"):
            parts = content.split(" ", 1)
            target = parts[0][1:]  # 去掉 @
            msg_text = parts[1] if len(parts) > 1 else ""
            if not msg_text:
                self.append_message("[系统] 私聊格式: @昵称 消息内容")
                return
            msg = {"type": TYPE_PRIVATE, "to": target, "content": msg_text}
            self.append_message(f"[私聊] 我 -> {target}: {msg_text}")
        else:
            msg = {"type": TYPE_BROADCAST, "content": content}
            self.append_message(f"[公聊] 我: {content}")

        try:
            self.sock.sendall(pack_message(msg))
        except OSError:
            self.append_message("[系统] 发送失败，连接已断开")
            self.set_disconnected()

    def receive_loop(self):
        """接收线程：循环 recv 并解析消息放入队列。"""
        buf = bytearray()
        while self.running:
            try:
                data = self.sock.recv(4096)
            except (ConnectionResetError, ConnectionAbortedError, OSError):
                data = b""
            if not data:
                self.msg_queue.put({"type": TYPE_DISCONNECTED})
                break
            buf.extend(data)
            for msg in unpack_from_buffer(buf):
                self.msg_queue.put(msg)

    def poll_queue(self):
        """定时轮询队列，处理收到的消息。"""
        try:
            while True:
                msg = self.msg_queue.get_nowait()
                self.handle_message(msg)
        except queue.Empty:
            pass
        if self.running:
            self.root.after(100, self.poll_queue)

    def handle_message(self, msg):
        msg_type = msg.get("type", "")

        if msg_type == TYPE_BROADCAST:
            self.append_message(f"[公聊] {msg['from']}: {msg['content']}")

        elif msg_type == TYPE_PRIVATE:
            self.append_message(f"[私聊] {msg['from']} -> 我: {msg['content']}")

        elif msg_type == TYPE_USER_ONLINE:
            nickname = msg["nickname"]
            self.user_listbox.insert(tk.END, nickname)
            self.append_message(f"[系统] {nickname} 上线了")

        elif msg_type == TYPE_USER_OFFLINE:
            nickname = msg["nickname"]
            self._remove_user_from_list(nickname)
            self.append_message(f"[系统] {nickname} 下线了")

        elif msg_type == TYPE_ERROR:
            self.append_message(f"[系统] 错误: {msg.get('reason', '')}")

        elif msg_type == TYPE_DISCONNECTED:
            self.append_message("[系统] 与服务器断开连接")
            self.set_disconnected()

    def _remove_user_from_list(self, nickname):
        """从在线列表框中移除指定用户。"""
        for i in range(self.user_listbox.size()):
            if self.user_listbox.get(i) == nickname:
                self.user_listbox.delete(i)
                break

    def set_disconnected(self):
        """设置断开连接状态。"""
        self.running = False
        self.input_entry.config(state=tk.DISABLED)
        self.append_message("[系统] 已断开连接")

    def do_close(self):
        """窗口关闭时的处理。"""
        self.running = False
        try:
            self.sock.sendall(pack_message({"type": TYPE_LOGOUT}))
        except OSError:
            pass
        try:
            self.sock.close()
        except OSError:
            pass
        self.root.destroy()
        self.on_close()


def main():
    login = LoginWindow()
    login.run()


if __name__ == "__main__":
    main()
```

```python
"""集成测试脚本"""
import socket
import time
import json
import sys
sys.path.insert(0, "lanchat")
from protocol import pack_message, unpack_from_buffer

def recv_until(sock, timeout=2):
    sock.settimeout(timeout)
    buf = bytearray()
    try:
        while True:
            data = sock.recv(4096)
            if not data:
                break
            buf.extend(data)
    except socket.timeout:
        pass
    return unpack_from_buffer(buf)

# 客户端 A 登录
a = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
a.connect(("127.0.0.1", 9999))
a.sendall(pack_message({"type": "login", "nickname": "Alice"}))
msgs = recv_until(a)
assert msgs[0]["type"] == "login_ok", f"login_ok expected, got {msgs[0]}"
print("[PASS] Alice 登录成功，在线用户:", msgs[0]["users"])

# 客户端 B 登录
b = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
b.connect(("127.0.0.1", 9999))
b.sendall(pack_message({"type": "login", "nickname": "Bob"}))
msgs = recv_until(b)
assert msgs[0]["type"] == "login_ok"
print("[PASS] Bob 登录成功")

# A 应收到 Bob 上线通知
msgs = recv_until(a)
onlines = [m for m in msgs if m["type"] == "user_online"]
assert any(m["nickname"] == "Bob" for m in onlines), "A 应收到 Bob 上线通知"
print("[PASS] Alice 收到 Bob 上线通知")

# A 发送公聊
a.sendall(pack_message({"type": "broadcast", "content": "Hello everyone!"}))

offlines = [m for m in msgs if m["type"] == "user_offline"]
assert any(m["nickname"] == "Bob" for m in offlines)
print("[PASS] Alice 收到 Bob 下线通知")
```
